uname -a
cat /etc/os-release
ubus call system board
ip -brief address
ip route show
ip rule show
cat /etc/resolv.conf
uci show network
uci show dhcp
uci show firewall
uci show wireless
logread -e netifd
logread -e dnsmasq
ss -lntup
